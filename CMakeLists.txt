cmake_minimum_required(VERSION 3.25)
project(tide-motion VERSION 0.5.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(FetchContent)
option(MOTION_VULKAN "Build native Kimodo GPU acceleration (requires Vulkan SDK at build time)" OFF)
set(MOTION_ORT_ROOT "" CACHE PATH "Existing ONNX Runtime CPU distribution")
# The patch exposes the pinned revision's conditioned sampler publicly (pose
# constraints over one prompt or a sequence); see notes/implementation.md.
FetchContent_Declare(kimodo GIT_REPOSITORY https://github.com/localai-org/kimodo.cpp.git
  GIT_TAG ed8a931aedb132f768190f6931c365855caba74c GIT_SUBMODULES ggml SOURCE_SUBDIR _unused
  PATCH_COMMAND "${CMAKE_COMMAND}" "-DPATCH=${CMAKE_CURRENT_SOURCE_DIR}/patches/kimodo-conditioning.patch"
    -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/apply-patch.cmake")
FetchContent_MakeAvailable(kimodo)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GGML_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN ${MOTION_VULKAN} CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CCACHE OFF CACHE BOOL "" FORCE)
set(GGML_ALL_WARNINGS OFF CACHE BOOL "" FORCE)
add_subdirectory("${kimodo_SOURCE_DIR}/ggml" "${CMAKE_BINARY_DIR}/ggml" EXCLUDE_FROM_ALL)
set(kimodo_sources capi model gguf diffusion motion_rep motion_decode ggml_weights denoiser sequence llm_tokenizer llm_text_encoder)
foreach(source IN LISTS kimodo_sources)
  list(APPEND native_sources "${kimodo_SOURCE_DIR}/src/${source}.cpp")
endforeach()
add_library(motion-kimodo STATIC ${native_sources})
target_include_directories(motion-kimodo PUBLIC "${kimodo_SOURCE_DIR}/include" PRIVATE "${kimodo_SOURCE_DIR}/src")
target_compile_definitions(motion-kimodo PRIVATE KIMODO_BUILD KIMODO_HAVE_GGML=1)
target_link_libraries(motion-kimodo PRIVATE ggml)
if(MOTION_VULKAN)
  target_compile_definitions(motion-kimodo PRIVATE KIMODO_HAVE_GGML_VULKAN=1)
  target_link_libraries(motion-kimodo PRIVATE ggml-vulkan)
endif()
set(JSON_Install OFF CACHE BOOL "" FORCE)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz
  URL_HASH SHA256=42f6e95cad6ec532fd372391373363b62a14af6d771056dbfc86160e6dfff7aa)
FetchContent_MakeAvailable(json)
FetchContent_Declare(picosha2 GIT_REPOSITORY https://github.com/okdshin/PicoSHA2.git GIT_TAG 161cb3fc4170fa7a3eca9e582cebd27cc4d1fe29 SOURCE_SUBDIR _unused)
FetchContent_MakeAvailable(picosha2)
FetchContent_Declare(cgltf GIT_REPOSITORY https://github.com/jkuhlmann/cgltf.git GIT_TAG 360db1a95480fe102ae9c69b27c5d101167ff5ba SOURCE_SUBDIR _unused)
FetchContent_MakeAvailable(cgltf)
# FBX characters and clips: the ufbx revision Tide's compiler vendors (v0.23.1), read with
# its load options so node names and transforms match what Tide imports.
FetchContent_Declare(ufbx GIT_REPOSITORY https://github.com/ufbx/ufbx.git GIT_TAG 26a482ae66871d7de36eb722aa060bce95bce274 SOURCE_SUBDIR _unused)
FetchContent_MakeAvailable(ufbx)
add_library(motion-ufbx STATIC "${ufbx_SOURCE_DIR}/ufbx.c")
target_include_directories(motion-ufbx PUBLIC "${ufbx_SOURCE_DIR}")
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(motion-ufbx PRIVATE -Wno-unused-function)
endif()
if(MSVC)
  target_compile_definitions(motion-ufbx PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  target_link_libraries(motion-ufbx PUBLIC m)
endif()
if(NOT MOTION_ORT_ROOT)
  if(WIN32 AND CMAKE_SYSTEM_PROCESSOR MATCHES "AMD64|amd64|x86_64")
    set(ort_platform win-x64)
    set(ort_extension zip)
    set(ort_hash 5869e26bb25314375f2d5ef1877e3ea06b1543d6c710e6e4a1ec2f7177ab4b2c)
  elseif(APPLE)
    if(CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
      set(ort_platform osx-arm64)
      set(ort_hash 2e1344a1926ee50eb25b38153cb6992c3567515aef0e677a1a520a1798896d6f)
    else()
      message(FATAL_ERROR "Supply MOTION_ORT_ROOT for an Intel Mac; the pinned ONNX release has no Intel Mac archive")
    endif()
    set(ort_extension tgz)
  elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64")
    set(ort_platform linux-x64)
    set(ort_extension tgz)
    set(ort_hash c849e36a1ef3e22d4ff217b9d1ce4a3cf5b9467b975610913ae959ad5b16bdc9)
  elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
    set(ort_platform linux-aarch64)
    set(ort_extension tgz)
    set(ort_hash 5317505e2da99be65f8e2775ba3c2afd6d2961fd78aba1bd8f56159f48820c78)
  else()
    message(FATAL_ERROR "Supply MOTION_ORT_ROOT for this host architecture")
  endif()
  FetchContent_Declare(ort URL "https://github.com/microsoft/onnxruntime/releases/download/v1.31.1/onnxruntime-${ort_platform}-1.31.1.${ort_extension}" URL_HASH SHA256=${ort_hash})
  FetchContent_MakeAvailable(ort)
  set(MOTION_ORT_ROOT "${ort_SOURCE_DIR}")
endif()
add_library(motion-ort SHARED IMPORTED)
set_target_properties(motion-ort PROPERTIES INTERFACE_INCLUDE_DIRECTORIES "${MOTION_ORT_ROOT}/include")
if(WIN32)
  set_target_properties(motion-ort PROPERTIES IMPORTED_LOCATION "${MOTION_ORT_ROOT}/lib/onnxruntime.dll" IMPORTED_IMPLIB "${MOTION_ORT_ROOT}/lib/onnxruntime.lib")
elseif(APPLE)
  set_target_properties(motion-ort PROPERTIES IMPORTED_LOCATION "${MOTION_ORT_ROOT}/lib/libonnxruntime.dylib")
else()
  set_target_properties(motion-ort PROPERTIES IMPORTED_LOCATION "${MOTION_ORT_ROOT}/lib/libonnxruntime.so")
endif()
add_library(motion-core STATIC src/motion.cpp src/gltf.cpp src/retarget.cpp src/bricks.cpp src/platform.cpp src/constraints.cpp src/fbx.cpp src/mannequin.cpp src/edit.cpp src/author.cpp)
target_include_directories(motion-core PUBLIC src PRIVATE "${cgltf_SOURCE_DIR}" "${kimodo_SOURCE_DIR}/src")
target_link_libraries(motion-core PUBLIC nlohmann_json::nlohmann_json motion-ort PRIVATE motion-ufbx)
add_executable(tide-motion src/main.cpp src/models.cpp)
if(WIN32)
  target_sources(tide-motion PRIVATE src/windows.manifest)
  target_link_libraries(tide-motion PRIVATE bcrypt)
endif()
target_link_libraries(tide-motion PRIVATE motion-core motion-kimodo)
if(MOTION_VULKAN)
  # The CLI probes Vulkan before Kimodo loads and falls back to the CPU. On Windows the
  # loader is delay-loaded, so a machine without vulkan-1.dll still starts.
  target_include_directories(tide-motion PRIVATE "${kimodo_SOURCE_DIR}/ggml/include")
  target_link_libraries(tide-motion PRIVATE ggml-vulkan)
  target_compile_definitions(tide-motion PRIVATE MOTION_VULKAN=1)
  if(MSVC)
    # GGML's C functions throw for an unusable driver; /EHsc would drop the catch.
    target_compile_options(tide-motion PRIVATE /EHc-)
  endif()
  if(WIN32)
    target_link_options(tide-motion PRIVATE /DELAYLOAD:vulkan-1.dll)
    target_link_libraries(tide-motion PRIVATE delayimp)
  endif()
endif()
target_include_directories(tide-motion PRIVATE "${picosha2_SOURCE_DIR}")
target_compile_definitions(tide-motion PRIVATE MOTION_VERSION="${PROJECT_VERSION}")
if(MSVC)
  target_compile_options(motion-core PRIVATE /W4 /WX)
  target_compile_options(tide-motion PRIVATE /W4 /WX)
  target_compile_definitions(motion-core PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
  target_compile_definitions(tide-motion PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
  target_compile_definitions(motion-kimodo PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
else()
  target_compile_options(motion-core PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_compile_options(tide-motion PRIVATE -Wall -Wextra -Wpedantic -Werror)
  set(CMAKE_INSTALL_RPATH "$ORIGIN;@loader_path")
endif()
add_custom_command(TARGET tide-motion POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:motion-ort>" "$<TARGET_FILE_DIR:tide-motion>")
install(TARGETS tide-motion RUNTIME DESTINATION bin COMPONENT runtime)
file(GLOB ort_runtime_files "${MOTION_ORT_ROOT}/lib/*.dll" "${MOTION_ORT_ROOT}/lib/libonnxruntime.so*" "${MOTION_ORT_ROOT}/lib/libonnxruntime*.dylib")
install(FILES ${ort_runtime_files} DESTINATION bin COMPONENT runtime)
if(APPLE)
  set_target_properties(tide-motion PROPERTIES INSTALL_RPATH "@loader_path")
elseif(UNIX)
  set_target_properties(tide-motion PROPERTIES INSTALL_RPATH "$ORIGIN")
endif()
install(DIRECTORY licenses/ DESTINATION licenses COMPONENT runtime)
install(FILES README.md PROMPTING.md LICENSE install.ps1 install.sh DESTINATION . COMPONENT runtime)
install(FILES notes/implementation.md DESTINATION notes COMPONENT runtime)
install(DIRECTORY examples/ DESTINATION examples COMPONENT runtime)
set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION bin)
set(CMAKE_INSTALL_SYSTEM_RUNTIME_COMPONENT runtime)
include(InstallRequiredSystemLibraries)
include(CTest)
if(BUILD_TESTING)
  # Kimodo's decoder checks the native pose encoder without loading a model.
  add_executable(motion-tests tests/tests.cpp "${kimodo_SOURCE_DIR}/src/motion_decode.cpp")
  target_include_directories(motion-tests PRIVATE "${kimodo_SOURCE_DIR}/src")
  target_link_libraries(motion-tests PRIVATE motion-core)
  add_custom_command(TARGET motion-tests POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:motion-ort>" "$<TARGET_FILE_DIR:motion-tests>")
  add_test(NAME motion-contracts COMMAND motion-tests)
  add_executable(retarget-tests tests/retarget.cpp)
  target_link_libraries(retarget-tests PRIVATE motion-core)
  add_custom_command(TARGET retarget-tests POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:motion-ort>" "$<TARGET_FILE_DIR:retarget-tests>")
  add_test(NAME retarget-contracts COMMAND retarget-tests)
  add_test(NAME cli-help COMMAND tide-motion --help)
  add_test(NAME cli-contracts COMMAND "${CMAKE_COMMAND}" "-DTOOL=$<TARGET_FILE:tide-motion>" "-DFIXTURE=$<TARGET_FILE:motion-tests>" "-DWORK=${CMAKE_CURRENT_BINARY_DIR}/cli-contracts" -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.cmake")
  add_test(NAME retarget-cli COMMAND "${CMAKE_COMMAND}" "-DTOOL=$<TARGET_FILE:tide-motion>" "-DFIXTURE=$<TARGET_FILE:retarget-tests>" "-DWORK=${CMAKE_CURRENT_BINARY_DIR}/retarget-cli" -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/retarget.cmake")
endif()
if(CMAKE_SYSTEM_PROCESSOR MATCHES "AMD64|amd64|x86_64")
  set(motion_arch x64)
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
  set(motion_arch arm64)
else()
  message(FATAL_ERROR "Unsupported package architecture: ${CMAKE_SYSTEM_PROCESSOR}")
endif()
if(WIN32)
  set(CPACK_GENERATOR ZIP)
  set(motion_os windows)
elseif(APPLE)
  set(CPACK_GENERATOR TGZ)
  set(motion_os macos)
else()
  set(CPACK_GENERATOR TGZ)
  set(motion_os linux)
endif()
set(CPACK_COMPONENTS_ALL runtime)
set(CPACK_ARCHIVE_COMPONENT_INSTALL ON)
set(CPACK_PACKAGE_FILE_NAME "tide-motion-${PROJECT_VERSION}-${motion_os}-${motion_arch}")
set(CPACK_ARCHIVE_RUNTIME_FILE_NAME "${CPACK_PACKAGE_FILE_NAME}")
include(CPack)
