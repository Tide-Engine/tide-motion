cmake_minimum_required(VERSION 3.25)
project(dart-source-probe LANGUAGES CXX)
enable_testing()
add_executable(dart-source-probe probe.cpp)
target_compile_features(dart-source-probe PRIVATE cxx_std_23)
if(MSVC)
  target_compile_options(dart-source-probe PRIVATE /W4 /WX)
else()
  target_compile_options(dart-source-probe PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME dart-source-math COMMAND dart-source-probe)
set(MOTION_ORT_ROOT "" CACHE PATH "Optional existing pinned ONNX Runtime 1.31.1 distribution")
if(MOTION_ORT_ROOT AND WIN32)
  add_executable(dart-ort-probe ort.cpp)
  target_compile_features(dart-ort-probe PRIVATE cxx_std_23)
  target_include_directories(dart-ort-probe PRIVATE "${MOTION_ORT_ROOT}/include")
  target_link_libraries(dart-ort-probe PRIVATE "${MOTION_ORT_ROOT}/lib/onnxruntime.lib")
  add_custom_command(TARGET dart-ort-probe POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${MOTION_ORT_ROOT}/lib/onnxruntime.dll" "$<TARGET_FILE_DIR:dart-ort-probe>")
  add_test(NAME dart-ort-operators COMMAND dart-ort-probe)
endif()
