cmake_minimum_required(VERSION 3.25)
project(ardy_native_probe LANGUAGES CXX)
include(FetchContent)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz
  URL_HASH SHA256=42f6e95cad6ec532fd372391373363b62a14af6d771056dbfc86160e6dfff7aa)
FetchContent_MakeAvailable(json)
add_executable(ardy-native-probe probe.cpp)
target_compile_features(ardy-native-probe PRIVATE cxx_std_23)
target_link_libraries(ardy-native-probe PRIVATE nlohmann_json::nlohmann_json)
if(MSVC)
  target_compile_options(ardy-native-probe PRIVATE /W4 /WX)
else()
  target_compile_options(ardy-native-probe PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
